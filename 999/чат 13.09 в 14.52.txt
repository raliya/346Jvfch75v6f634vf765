 private static async Task<bool>
 Build10RetryIspTxtAsync(
 HttpClient client,
 string panelUrl,
 string sessionId,
 string domain,
 CancellationToken cancellationToken)
 {
 var openFields =
 new Dictionary<string, string>
 {
 ["out"] = "devel",
 ["lang"] = "ru",
 ["auth"] = sessionId,
 ["func"] = "webdomain.letsencrypt.txt",
 ["elid"] = domain,
 ["elname"] = domain
 };

 var form =
 await Build10PostIspAsync(
 client,
 Build10NormalizeIspUrl(panelUrl),
 openFields,
 cancellationToken);

 form.Descendants().Any(element =>
 string.Equals(
 element.Name.LocalName,
 "field",
 StringComparison.OrdinalIgnoreCase) &&
 string.Equals(
 element.Attribute("name")?.Value,
 "txt_found_banner",
 StringComparison.OrdinalIgnoreCase));

 // возвращённый формой, например owner%#%domain_le1.

 element.Name.LocalName,

 requestId.Length > 512 ||
 requestId.Any(char.IsControl) ||

 StringComparison.Ordinal))

 "ISPmanager не вернул допустимый идентификатор " +
 "Wildcard-заявки. Повторная команда не отправлена.");

 new Dictionary<string, string>

 ["out"] = "xml",
 ["lang"] = "ru",
 ["auth"] = sessionId,
 ["func"] = "webdomain.letsencrypt.txt",
 ["elid"] = requestId,
 ["elname"] = domain,
 ["tconvert"] = "punycode",
 ["sok"] = "ok"

 client,
 Build10NormalizeIspUrl(panelUrl),
 submitFields,
 cancellationToken);

    function Find-ExactlyOnce {
        param(
            [Parameter(Mandatory = $true)]
            [string] $Text,

            [Parameter(Mandatory = $true)]
            [string] $Needle,

            [Parameter(Mandatory = $true)]
            [string] $Description
        )

        $first = $Text.IndexOf(
            $Needle,
            [StringComparison]::Ordinal
        )

        if ($first -lt 0) {
            throw (
                'Не найден участок: ' +
                $Description +
                '.'
            )
        }

        $second = $Text.IndexOf(
            $Needle,
            $first + $Needle.Length,
            [StringComparison]::Ordinal
        )

        if ($second -ge 0) {
            throw (
                'Участок найден несколько раз: ' +
                $Description +
                '.'
            )
        }

        return $first
    }

    function Replace-LiteralOnce {
        param(
            [Parameter(Mandatory = $true)]
            [string] $Text,

            [Parameter(Mandatory = $true)]
            [string] $OldText,

            [Parameter(Mandatory = $true)]
            [AllowEmptyString()]
            [string] $NewText,

            [Parameter(Mandatory = $true)]
            [string] $Description

            -Text $Text `
            -Needle $OldText `
            -Description $Description

        return (
            $Text.Substring(0, $position) +
            $NewText +
            $Text.Substring($position + $OldText.Length)

function Write-Step {
    param([Parameter(Mandatory)][string]$Text)

    Write-Host ''
    Write-Host ('=== ' + $Text + ' ===') -ForegroundColor Cyan
}

function Require-Command {
    param([Parameter(Mandatory)][string]$Name)

    if (-not (Get-Command $Name -ErrorAction SilentlyContinue)) {
        throw "Не найдена команда '$Name'. Установите .NET 8 SDK."
    }
}

function Replace-Exact {
    param(
        [Parameter(Mandatory)][string]$Text,
        [Parameter(Mandatory)][string]$Old,
        [Parameter(Mandatory)][string]$New,
        [Parameter(Mandatory)][string]$Description
    )

    $count = (
        [regex]::Matches(
            $Text,
            [regex]::Escape($Old)
        )
    ).Count

    if ($count -ne 1) {
        throw (
            "$Description: ожидалось одно совпадение, найдено $count. " +
            'Исходники изменились, автоматическая правка остановлена.'
        )
    }

    return $Text.Replace(
        $Old,
        $New,
        [StringComparison]::Ordinal
    )
}

function Copy-IfExists {
    param(
        [Parameter(Mandatory)][string]$Source,
        [Parameter(Mandatory)][string]$Destination
    )

    if (Test-Path -LiteralPath $Source) {
        Copy-Item `
            -LiteralPath $Source `
            -Destination $Destination `
            -Recurse `
            -Force
    }

        {
            const string message =
                "Повторная проверка TXT остановлена.";

            SetBuild10Status(
                _build10ProgressBar?.Value ?? 0,
                message);

            AddLog(
                "WARN",
                string.Empty,
                message);
        }
        catch (Exception exception)
        {
            var message =
                SensitiveDataRedactor.Redact(
                    exception.Message);

            SetBuild10Status(
                _build10ProgressBar?.Value ?? 0,
                message);

            AddLog(
                "ERROR",
                string.Empty,
                message);

            MessageBox.Show(
                this,
                message,
                "Ошибка повторной проверки TXT",
                MessageBoxButton.OK,
                MessageBoxImage.Error);
        }
        finally
        {
            _cancellationTokenSource?.Dispose();
            _cancellationTokenSource = null;
            _isRunning = false;

            if (_wildcardRetryTxtButton is not null)
            {
                _wildcardRetryTxtButton.IsEnabled = true;
            }

            if (_build10SyncButton is not null)
            {
                _build10SyncButton.IsEnabled = true;
            }

            StartButton.IsEnabled = true;
            CancelButton.IsEnabled = false;
        }
    }

    private static Dictionary<string, string>
        WildcardReadFormValues(
            XDocument document)
    {
        var result =
            new Dictionary<string, string>(
                StringComparer.Ordinal);

        var root = document.Root ??
            throw new InvalidOperationException(
                "ISPmanager вернул пустую форму TXT.");

        var reserved =
            new HashSet<string>(
                StringComparer.OrdinalIgnoreCase)
            {
                "auth",
                "authinfo",
                "func",
                "out",
                "lang",
                "sok",
                "password",
                "confirm"
            };

        var controls = root
            .Elements("metadata")
            .SelectMany(element =>
                element.Descendants())
            .Where(element =>
                element.Name.LocalName is
                    "input" or
                    "select" or
                    "textarea" or
                    "slider");

        foreach (var control in controls)
        {
            var name =
                control.Attribute("name")?
                    .Value?
                    .Trim() ??
                string.Empty;

            var type =
                control.Attribute("type")?
                    .Value?
                    .Trim() ??
                string.Empty;

            if (name.Length == 0 ||
                reserved.Contains(name) ||
                name.Contains(
                    "pass",
                    StringComparison.OrdinalIgnoreCase) ||
                type.Equals(
                    "password",
                    StringComparison.OrdinalIgnoreCase))
            {
                continue;
            }

            var valueElement = root
                .Elements()
                .FirstOrDefault(element =>
                    string.Equals(
                        element.Name.LocalName,
                        name,
                        StringComparison.Ordinal));

            if (valueElement is not null)
            {
                result[name] =
                    valueElement.Value;
            }
        }

        return result;
    }

    private static string WildcardXmlValue(
        XDocument document,
        string name)
    {
        return document.Root?
            .Elements()
            .FirstOrDefault(element =>
                string.Equals(
                    element.Name.LocalName,
                    name,
                    StringComparison.OrdinalIgnoreCase))
            ?.Value ??
            string.Empty;
    }
