  493:     private static string ReadPhpValue(
  494:         string configText,
  495:         string key)
  496:     {
  497:         var pattern =
  498:             @"[""']" +
  499:             Regex.Escape(key) +
  500:             @"[""']\s*=>\s*" +
  501:             @"(?<quote>[""'])" +
  502:             @"(?<value>(?:\\.|(?!\k<quote>).)*)" +
  503:             @"\k<quote>";
  504: 
  505:         var match = Regex.Match(
  506:             configText,
  507:             pattern,
  508:             RegexOptions.CultureInvariant |
  509:             RegexOptions.Singleline);
  510: 
  511:         if (!match.Success)
  512:         {
  513:             throw new InvalidOperationException(
  514:                 $"В config.php не найден параметр '{key}'.");
  515:         }
  516: 
  517:         var value = match
  518:             .Groups["value"]
  519:             .Value;
  520: 
  521:         value = Regex.Replace(
  522:             value,
  523:             @"\\(['""\\])",
  524:             "$1");
  525: 
  526:         if (string.IsNullOrWhiteSpace(value))
  527:         {
  528:             throw new InvalidOperationException(
  529:                 $"Параметр '{key}' в config.php пуст.");
  530:         }
  531: 
  532:         return value;
  533:     }
