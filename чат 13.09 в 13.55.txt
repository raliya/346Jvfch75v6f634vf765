            $Replacement +

    private static async Task<bool>
        Build10RetryIspTxtAsync(
            HttpClient client,
            string panelUrl,
            string sessionId,
            string domain,
            CancellationToken cancellationToken)
    {
        var openFields =
            new Dictionary<string, string>
            {
                ["out"] = "devel",
                ["lang"] = "ru",
                ["auth"] = sessionId,
                ["func"] = "webdomain.letsencrypt.txt",
                ["elid"] = domain,
                ["elname"] = domain
            };

        var form =
            await Build10PostIspAsync(
                client,
                Build10NormalizeIspUrl(panelUrl),
                openFields,
                cancellationToken);

        var alreadyFound =
            form.Descendants().Any(element =>
                string.Equals(
                    element.Name.LocalName,
                    "field",
                    StringComparison.OrdinalIgnoreCase) &&
                string.Equals(
                    element.Attribute("name")?.Value,
                    "txt_found_banner",
                    StringComparison.OrdinalIgnoreCase));

        if (alreadyFound)
            return true;

        // Важно: отправлять нужно идентификатор заявки,
        // возвращённый формой, например owner%#%domain_le1.
        // Передача самого домена приводила к ошибке поля txt.
        var requestId =
            form.Root?
                .Elements()
                .FirstOrDefault(element =>
                    string.Equals(
                        element.Name.LocalName,
                        "elid",
                        StringComparison.OrdinalIgnoreCase))
                ?.Value
                .Trim() ?? string.Empty;

        if (requestId.Length == 0 ||
            requestId.Length > 512 ||
            requestId.Any(char.IsControl) ||
            !requestId.Contains(
                "%#%",
                StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "ISPmanager не вернул допустимый идентификатор " +
                "Wildcard-заявки. Повторная команда не отправлена.");
        }

        var submitFields =
            new Dictionary<string, string>
            {
                ["out"] = "xml",
                ["lang"] = "ru",
                ["auth"] = sessionId,
                ["func"] = "webdomain.letsencrypt.txt",
                ["elid"] = requestId,
                ["elname"] = domain,
                ["tconvert"] = "punycode",
                ["sok"] = "ok"
            };

        await Build10PostIspAsync(
            client,
            Build10NormalizeIspUrl(panelUrl),
            submitFields,
